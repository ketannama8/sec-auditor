# Terminal colour output (cross-platform ANSI colour support)
# Used by: sec_auditor/reporter.py
# Purpose: Colour-codes port states and header findings in terminal output
colorama>=0.4.6

# Web framework for the browser-based UI
# Used by: sec_auditor/web.py + sec_auditor/templates/index.html
# Purpose: Serves the single-page audit UI and the /api/scan JSON endpoint
flask>=3.0

# argparse        — CLI argument parsing
# json            — JSON serialisation for report export
# dataclasses     — Typed result objects (PortScanResult, HeaderAuditResult)
# re              — URL scheme stripping regex
# os              — Directory creation for JSON export
# threading       — Background server thread
